    $$
    frac{text{sin}theta_1}{text{sin}theta_2} = frac{v_1}{v_2} = frac{sqrt{frac{E_1}{rho_1}}}{sqrt{frac{E_2}{rho_2}}}

    $$
    text{sin}theta_2 = frac{v_2}{v_1}text{sin}theta_1 = frac{sqrt{frac{E_2}{rho_2}}}{sqrt{frac{E_1}{rho_1}}}
    $$
